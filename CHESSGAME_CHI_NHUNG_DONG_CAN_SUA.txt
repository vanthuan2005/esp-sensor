A2. Trong Chess.java, THEM ham sau TRUOC ham getGame():

---------------------------
    public int[] getKingPosition(int side) {
        int king = side == 1 ? 12 : 6;

        for (int row = 0; row < SIDE; row++) {
            for (int col = 0; col < SIDE; col++) {
                if (game[row][col] == king) {
                    return new int[]{row, col};
                }
            }
        }

        return new int[]{-1, -1};
    }
