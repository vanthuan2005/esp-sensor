    public boolean hasAnyLegalMove(int side) {

        for (int row = 0; row < SIDE; row++) {
            for (int col = 0; col < SIDE; col++) {

                int p = game[row][col];

                if (p == 0) {
                    continue;
                }

                // side = 1 -> WHITE (7..12)
                if (side == 1 && (p < 7 || p > 12)) {
                    continue;
                }

                // side = 2 -> BLACK (1..6)
                if (side == 2 && (p < 1 || p > 6)) {
                    continue;
                }

                Piece testPiece = new Piece(row, col, p, game);
                int[][] moves = testPiece.possibleMoves();

                moves = filterLegalMoves(row, col, moves);

                for (int r = 0; r < SIDE; r++) {
                    for (int c = 0; c < SIDE; c++) {

                        // Khi dang bi chieu, cac nuoc thoat chieu
                        // thong thuong se la 2 (di) hoac 3 (an).
                        if (moves[r][c] == 2 || moves[r][c] == 3) {
                            return true;
                        }
                    }
                }
            }
        }

        return false;
    }


    public boolean isCheckmate(int side) {

        return isKingInCheck(side) && !hasAnyLegalMove(side);
    }
